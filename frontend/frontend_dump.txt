  const [error, setError] = useState<string | null>(null);

  const loadStats = useCallback(() => {
    api<Stats>('/admin/stats').then(setStats).catch((e: Error) => setError(e.message));
  }, []);

  useEffect(() => {
    loadStats();
    api<ScanState>('/admin/scan').then(setScan).catch(() => undefined);
  }, [loadStats]);

  // пока идёт сканирование — опрашиваем прогресс
  const running = scan?.running;
  useEffect(() => {
    if (!running) return;
    const t = setInterval(() => {
      api<ScanState>('/admin/scan')
        .then((s) => {
          setScan(s);
          if (!s.running) loadStats();
        })
        .catch(() => undefined);
    }, 1500);
    return () => clearInterval(t);
  }, [running, loadStats]);

  const startScan = async () => {
    setError(null);
    try {
      setScan(await api<ScanState>('/admin/scan', { method: 'POST', body: { pruneMissing: prune } }));
    } catch (e) {
      setError((e as Error).message);
    }
  };

  const enrich = async () => {
    try {
      await api('/admin/enrich', { method: 'POST' });

  };

  };

function ReviewRow({ track, onSaved }: { track: Track; onSaved: (id: string) => void }) {
  const [artist, setArtist] = useState(track.artist === 'Unknown Artist' ? '' : track.artist);
  const [title, setTitle] = useState(track.title);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const save = async () => {
    setBusy(true);
    setError(null);
    try {
      await api(`/admin/tracks/${track.id}`, { method: 'PATCH', body: { artist: artist.trim(), title: title.trim() } });
      onSaved(track.id);
    } catch (e) {
      setError((e as Error).message);
      setBusy(false);
    }
  };

  const remove = async () => {
    if (!confirm('Удалить трек из базы? Файл на диске останется.')) return;
    await api(`/admin/tracks/${track.id}`, { method: 'DELETE' });
    onSaved(track.id);
  };

  return (
    <div className="space-y-2 px-4 py-3">
      <div className="truncate text-xs text-zinc-500" title={track.filePath}>
        {track.owner ? `${track.owner} · ` : ''}{track.filePath?.split(/[\\/]/).pop()} · {formatDuration(track.duration)} · {formatBytes(track.fileSize ?? 0)}
      </div>
      <div className="flex flex-wrap gap-2">
        <input className="input !w-auto min-w-0 flex-1" placeholder="Исполнитель" value={artist} onChange={(e) => setArtist(e.target.value)} />
        <input className="input !w-auto min-w-0 flex-[2]" placeholder="Название" value={title} onChange={(e) => setTitle(e.target.value)} />
        <button className="btn btn-primary" onClick={save} disabled={busy || !artist.trim() || !title.trim()}>
          <Check size={16} /> Сохранить
        </button>
        <button className="btn" onClick={remove} title="Удалить из базы">
          <Trash2 size={16} />
        </button>
      </div>
      <Err msg={error} />
    </div>
  );
}

const SCOPES = 'playlist-read-private playlist-read-collaborative';
const TOKEN_KEY = 'loral-sp-token';
const VERIFIER_KEY = 'loral-sp-verifier';
const STATE_KEY = 'loral-sp-state';
const PENDING_KEY = 'loral-pending-import';

const ALPHABET = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-._~';

function randomString(len = 64): string {
  const bytes = crypto.getRandomValues(new Uint8Array(len));
  return Array.from(bytes, (b) => ALPHABET[b % ALPHABET.length]).join('');
}

async function challengeFor(verifier: string): Promise<string> {
  const digest = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(verifier));
  return btoa(String.fromCharCode(...new Uint8Array(digest)))
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/, '');
}

class SpotifyError extends Error {
  constructor(
    public status: number,
    message: string,
  ) {
    super(message);
  }
}

function loadToken(): string | null {
  try {
    const raw = sessionStorage.getItem(TOKEN_KEY);
    if (!raw) return null;
    const { token, exp } = JSON.parse(raw) as { token: string; exp: number };
    return Date.now() < exp ? token : null;
  } catch {
    return null;
  }
}

async function spotify<T>(token: string, urlOrPath: string): Promise<T> {
  const url = urlOrPath.startsWith('http') ? urlOrPath : `https://api.spotify.com/v1${urlOrPath}`;
  const res = await fetch(url, { headers: { Authorization: `Bearer ${token}` } });
  if (res.ok) return (await res.json()) as T;
  const body = (await res.json().catch(() => null)) as { error?: { message?: string } } | null;
  throw new SpotifyError(res.status, body?.error?.message ?? `Ошибка Spotify ${res.status}`);
}

interface Paging<T> {
  items: T[];
  next: string | null;
}

async function fetchAll<T>(token: string, firstUrl: string, limit = 5000): Promise<T[]> {
  const out: T[] = [];
  let next: string | null = firstUrl;
  while (next && out.length < limit) {
    const page: Paging<T> = await spotify<Paging<T>>(token, next);
    out.push(...page.items);
    next = page.next;
  }
  return out;
}

interface RawPlaylist {
  id: string;
  name: string;
  // Spotify переименовал tracks → items (февраль 2026); поддерживаем оба варианта
  items?: { total?: number };
  tracks?: { total?: number };
}
interface RawEntry {
  item?: RawTrack | null;
  track?: RawTrack | null;
}
interface RawTrack {
  name?: string;
  type?: string;
  artists?: { name: string }[];
}

async function fetchPlaylistTracks(token: string, id: string): Promise<ImportTrack[]> {
  let entries: RawEntry[];
  try {
    entries = await fetchAll<RawEntry>(token, `/playlists/${id}/items?limit=100&additional_types=track`);
  } catch (e) {
    if (!(e instanceof SpotifyError) || e.status !== 404) throw e;
    entries = await fetchAll<RawEntry>(token, `/playlists/${id}/tracks?limit=100`); // старое название эндпоинта
  }
  const out: ImportTrack[] = [];
  for (const e of entries) {
    const t = e.item ?? e.track;
    if (!t?.name || (t.type && t.type !== 'track')) continue; // пропускаем подкасты и пустые записи
    const artist = t.artists?.[0]?.name;
    if (artist) out.push({ artist, title: t.name });
  }
  return out;
}

          patch(id, { state: 'done', progress: 100, message: 'Добавлено в плейлист и библиотеку' });
        } else if (r.status === 'duplicate') {
          patch(id, { state: 'invalid', message: r.error ?? 'Уже есть в библиотеке' });
        } else {
          patch(id, { state: 'error', message: r.error ?? 'Не удалось добавить' });
        }
      } catch (e) {
        patch(id, { state: 'error', message: e instanceof ApiError ? e.message : 'Ошибка загрузки' });
      }
    }

    // Сбрасываем ID плейлиста для следующей сессии загрузки
    playlistIdRef.current = null;
    running.current = false;
  };

  const addFiles = (list: FileList | File[]) => {
    const fresh: Row[] = [];
    for (const file of Array.from(list)) {
      const id = ++rowId;
      const check = validateUploadName(file.name);
      if (check.ok) {
        fresh.push({ id, name: file.name, size: file.size, state: 'queued', progress: 0 });
        queue.current.push({ id, file });
      } else {
        fresh.push({ id, name: file.name, size: file.size, state: 'invalid', progress: 0, message: check.error });
      }
    }
    setRows((rs) => [...fresh, ...rs]);
    void pump();
  };

  if (!canUpload) {
    return <div className="card p-8 text-center text-sm text-pink-300/70">У вас нет прав на загрузку треков. Обратитесь к администратору.</div>;

        <div className="text-xs text-pink-300/60">
          Формат имени: <code className="rounded-md bg-pink-950/60 px-2 py-0.5 border border-pink-500/20 text-pink-200">Исполнитель - Название.mp3</code>
        </div>

          className="hidden"
          onChange={(e) => {

              {(r.state === 'invalid' || r.state === 'error') && <AlertCircle size={18} className="shrink-0 text-rose-400" />}
              {r.state === 'uploading' && <Loader2 size={18} className="shrink-0 animate-spin text-pink-300" />}
              {r.state === 'queued' && <Loader2 size={18} className="shrink-0 text-pink-500/40" />}
              <div className="min-w-0 flex-1">
                <div className="truncate text-sm font-medium text-pink-100">{r.name}</div>
                <div className={cn('text-xs', r.state === 'invalid' || r.state === 'error' ? 'text-rose-300' : 'text-pink-300/60')}>
                  {r.state === 'uploading'

}

function onPlayError(e: unknown) {
  // AbortError — просто пришёл новый трек; остальное (например, запрет автовоспроизведения) — останавливаем
  if ((e as Error)?.name !== 'AbortError') usePlayer.getState()._setPlaying(false);
}

/**
 * Единственный <audio> приложения. Компонент смонтирован в layout группы (app),
 * поэтому переходы между страницами его не размонтируют и музыка не прерывается.
 */
export function GlobalPlayer() {
  const queue = usePlayer((s) => s.queue);
  const index = usePlayer((s) => s.index);
  const playing = usePlayer((s) => s.playing);
  const shuffle = usePlayer((s) => s.shuffle);
  const repeat = usePlayer((s) => s.repeat);
  const volume = usePlayer((s) => s.volume);
  const muted = usePlayer((s) => s.muted);
  const currentTime = usePlayer((s) => s.currentTime);
  const duration = usePlayer((s) => s.duration);
  const queueOpen = usePlayer((s) => s.queueOpen);
  const error = usePlayer((s) => s.error);
  const p = usePlayer.getState; // действия берём из стора напрямую

  const current = queue[index];
  const switching = useRef(false);
  const retried = useRef<string | null>(null);
  const [scrub, setScrub] = useState<number | null>(null);

  // 1) Смена трека: подставляем источник и запускаем, если пользователь хочет играть
  useEffect(() => {
    const a = audioRef.current;
    if (!a) return;
    if (!current) {
      a.pause();
      a.removeAttribute('src');
      a.load();
      return;
    }
    switching.current = true;
    a.src = current.kind === 'local' ? streamUrl(current.id!) : current.src!;
    if (usePlayer.getState().playing) a.play().catch(onPlayError);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [current?.key]);

  // 2) Пауза / воспроизведение из UI
  useEffect(() => {
    const a = audioRef.current;
    if (!a || !current) return;
    if (playing && a.paused) a.play().catch(onPlayError);
    else if (!playing && !a.paused) a.pause();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [playing]);

  // 3) Громкость
  useEffect(() => {
    const a = audioRef.current;
    if (a) {
      a.volume = volume;
      a.muted = muted;
    }
  }, [volume, muted]);

  // 4) Медиа-клавиши и экран блокировки
  useEffect(() => {
    if (typeof navigator === 'undefined' || !('mediaSession' in navigator)) return;
    const ms = navigator.mediaSession;
    if (!current) {
      ms.metadata = null;
      return;
    }
    const cover = assetUrl(current.coverUrl);
    try {
      ms.metadata = new MediaMetadata({
        title: current.title,
        artist: current.artist,
        artwork: cover ? [{ src: new URL(cover, location.href).href }] : [],
      });
    } catch {
      /* не критично */
    }
    ms.setActionHandler('play', () => usePlayer.setState({ playing: true }));
    ms.setActionHandler('pause', () => usePlayer.setState({ playing: false }));
    ms.setActionHandler('previoustrack', () => p().prev());
    ms.setActionHandler('nexttrack', () => p().next());
    ms.setActionHandler('seekto', (d) => {
      if (d.seekTime != null) p().seek(d.seekTime);
    });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [current?.key]);

  useEffect(() => {
    if (typeof navigator !== 'undefined' && 'mediaSession' in navigator) {
      navigator.mediaSession.playbackState = playing ? 'playing' : 'paused';
    }
  }, [playing]);

  const handleError = async () => {
    const a = audioRef.current;
    const s = usePlayer.getState();
    const cur = s.queue[s.index];
    if (!a || !cur || !a.getAttribute('src')) return;
    // Частая причина для локальных треков — истёкший stream-токен: обновляем и пробуем один раз
    if (cur.kind === 'local' && retried.current !== cur.key) {
      retried.current = cur.key;
      const resumeAt = a.currentTime;
      try {
        await refreshStreamToken();
      } catch {
        /* попробуем со старым */
      }
      a.src = streamUrl(cur.id!);
      a.addEventListener('loadedmetadata', () => (a.currentTime = resumeAt), { once: true });
      if (s.playing) a.play().catch(onPlayError);
      return;
    }
    s._setError('Не удалось воспроизвести трек');
  };

  const dur = duration || current?.duration || 0;
  const shown = scrub ?? currentTime;
  const commitScrub = () => {
    if (scrub !== null) {
      p().seek(scrub);
      setScrub(null);
    }

'use client';

import { useEffect, useRef, useState } from 'react';
import { Check, Plus } from 'lucide-react';
import { api } from '@/lib/api';
import type { PlaylistSummary } from '@/lib/types';

/** Небольшое всплывающее окно «Добавить в плейлист». */
export function PlaylistPicker({ trackId, onClose }: { trackId: string; onClose: () => void }) {
  const [lists, setLists] = useState<PlaylistSummary[] | null>(null);
  const [name, setName] = useState('');
  const [done, setDone] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    api<{ items: PlaylistSummary[] }>('/playlists')
      .then((r) => setLists(r.items))
      .catch(() => setLists([]));
  }, []);

  useEffect(() => {
    const onDown = (e: MouseEvent) => {
      if (ref.current && !ref.current.contains(e.target as Node)) onClose();
    };
    document.addEventListener('mousedown', onDown);
    return () => document.removeEventListener('mousedown', onDown);
  }, [onClose]);

  const add = async (playlistId: string, label: string) => {
    try {
      await api(`/playlists/${playlistId}/tracks`, { method: 'POST', body: { trackId } });
      setDone(label);
      setTimeout(onClose, 700);
    } catch (e) {
      setError((e as Error).message);
    }
  };

  const create = async () => {
    const n = name.trim();
    if (!n) return;
    try {
      const p = await api<PlaylistSummary>('/playlists', { method: 'POST', body: { name: n } });
      await add(p.id, p.name);
    } catch (e) {
      setError((e as Error).message);
    }
  };

  return (
    <div ref={ref} className="card absolute right-0 top-full z-30 mt-1 w-64 p-2 text-sm shadow-2xl" onClick={(e) => e.stopPropagation()}>
      {done ? (
        <div className="flex items-center gap-2 px-2 py-3 text-accent">
          <Check size={16} /> Добавлено в «{done}»
        </div>
      ) : (
        <>
          <div className="max-h-48 overflow-y-auto">
            {lists === null && <div className="px-2 py-2 text-zinc-500">Загрузка…</div>}
            {lists?.length === 0 && <div className="px-2 py-2 text-zinc-500">Плейлистов пока нет</div>}
            {lists?.map((p) => (
              <button
                key={p.id}
                className="flex w-full items-center justify-between rounded px-2 py-1.5 text-left hover:bg-zinc-800"
                onClick={() => add(p.id, p.name)}
              >
                <span className="truncate">{p.name}</span>
                <span className="ml-2 text-xs text-zinc-500">{p.trackCount}</span>
              </button>
            ))}
          </div>
          <div className="mt-2 flex gap-1 border-t border-zinc-800 pt-2">
            <input
              className="input !py-1"
              placeholder="Новый плейлист"
              value={name}
              onChange={(e) => setName(e.target.value)}
              onKeyDown={(e) => e.key === 'Enter' && create()}
            />
            <button className="btn btn-primary !px-2" onClick={create} aria-label="Создать">
              <Plus size={16} />
            </button>
          </div>
          {error && <div className="mt-1 px-1 text-xs text-red-400">{error}</div>}
        </>
      )}
    </div>
  );
}

import { useAuth } from '@/store/auth';
import type { Session } from '@/lib/types';

export const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? '/api';

export class ApiError extends Error {
  constructor(
    public status: number,
    public code: string,
    message: string,
    public data?: Record<string, unknown>,
  ) {
    super(message);
  }
}

type Query = Record<string, string | number | boolean | null | undefined>;

function buildUrl(path: string, query?: Query): string {
  const qs = new URLSearchParams();
  for (const [k, v] of Object.entries(query ?? {})) {
    if (v !== undefined && v !== null && v !== '') qs.set(k, String(v));
  }
  const s = qs.toString();
  return `${API_BASE}${path}${s ? `?${s}` : ''}`;
}

/** Обложки отдаются бэкендом как /api/covers/…; при прямом API_URL подставляем его origin. */
export function assetUrl(url: string | null | undefined): string | undefined {
  if (!url) return undefined;
  if (/^https?:\/\//.test(url)) return url;
  if (API_BASE.startsWith('http')) return API_BASE.replace(/\/api\/?$/, '') + url;
  return url;
}

export function streamUrl(trackId: string): string {
  const token = useAuth.getState().streamToken ?? '';
  return `${API_BASE}/tracks/${trackId}/stream?token=${encodeURIComponent(token)}`;
}

let refreshing: Promise<boolean> | null = null;

/** Обновляет пару токенов по refresh-токену (один запрос на всех параллельных вызовов). */
export function refreshSession(): Promise<boolean> {
  refreshing ??= (async () => {
    const { refreshToken, setSession, logout } = useAuth.getState();
    if (!refreshToken) return false;
    try {
      const res = await fetch(`${API_BASE}/auth/refresh`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ refreshToken }),
      });
      if (!res.ok) {
        if (res.status === 401 || res.status === 403) logout();
        return false;
      }
      setSession((await res.json()) as Session);
      return true;
    } catch {
      return false;
    }
  })().finally(() => {
    refreshing = null;
  });
  return refreshing;
}

export async function refreshStreamToken(): Promise<void> {
  const r = await api<{ streamToken: string }>('/auth/stream-token');
  useAuth.getState().setStreamToken(r.streamToken);
}

interface ApiOptions {
  method?: 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE';
  body?: unknown;
  query?: Query;
  signal?: AbortSignal;
  /** не пытаться обновлять сессию (для /auth/login и т.п.) */
  anonymous?: boolean;
}

export async function api<T = unknown>(path: string, opts: ApiOptions = {}): Promise<T> {
  const doFetch = () => {
    const token = useAuth.getState().accessToken;
    return fetch(buildUrl(path, opts.query), {
      method: opts.method ?? 'GET',
      headers: {
        ...(opts.body !== undefined ? { 'Content-Type': 'application/json' } : {}),
        ...(token && !opts.anonymous ? { Authorization: `Bearer ${token}` } : {}),
      },
      body: opts.body !== undefined ? JSON.stringify(opts.body) : undefined,
      signal: opts.signal,
    });
  };

  let res: Response;
  try {
    res = await doFetch();
  } catch (e) {
    if ((e as Error).name === 'AbortError') throw e;
    throw new ApiError(0, 'network', 'Нет связи с сервером');
  }

  if (res.status === 401 && !opts.anonymous && useAuth.getState().refreshToken) {
    if (await refreshSession()) res = await doFetch();
  }
  if (res.status === 204) return undefined as T;

  const data = (await res.json().catch(() => null)) as (Record<string, unknown> & { error?: string; message?: string }) | null;
  if (!res.ok) {
    if (res.status === 401 && !opts.anonymous) useAuth.getState().logout();
    throw new ApiError(res.status, data?.error ?? 'error', data?.message ?? res.statusText, data ?? undefined);
  }
  return data as T;
}

export interface UploadOutcome {
  fileName: string;
  status: 'created' | 'duplicate' | 'error';
  error?: string;
}

/** 
 * Загрузка одного файла в личную медиатеку с прогрессом и повторами при ошибках 5xx / сбоях сети.
 */
export function uploadFile(file: File, onProgress: (pct: number) => void): Promise<UploadOutcome> {
  const maxRetries = 3;

  const sendOnce = (canRetrySession: boolean): Promise<UploadOutcome> => {
    return new Promise((resolve, reject) => {
      const xhr = new XMLHttpRequest();
      xhr.open('POST', `${API_BASE}/upload`);
      xhr.setRequestHeader('Authorization', `Bearer ${useAuth.getState().accessToken ?? ''}`);
      
      xhr.upload.onprogress = (e) => {
        if (e.lengthComputable) onProgress(Math.round((e.loaded / e.total) * 100));
      };

      xhr.onerror = () => reject(new ApiError(0, 'network', 'Нет связи с сервером'));
      
      xhr.onload = async () => {
        if (xhr.status === 401 && canRetrySession && (await refreshSession())) {
          return sendOnce(false).then(resolve, reject);
        }
        let data: { results?: UploadOutcome[]; message?: string; error?: string } = {};
        try {
          data = JSON.parse(xhr.responseText);
        } catch {
          /* пустой ответ */
        }
        if (xhr.status >= 400) {
          return reject(new ApiError(xhr.status, data.error ?? 'error', data.message ?? `Ошибка ${xhr.status}`));
        }
        resolve(data.results?.[0] ?? { fileName: file.name, status: 'error', error: 'Пустой ответ сервера' });
      };

      const fd = new FormData();
      fd.append('files', file, file.name);
      xhr.send(fd);
    });
  };

  return (async () => {
    for (let attempt = 0; attempt < maxRetries; attempt++) {
      try {
        return await sendOnce(true);
      } catch (err) {
        if (err instanceof ApiError) {
          // Если ошибка на стороне клиента (4xx, кроме 429), не повторяем
          if (err.status >= 400 && err.status < 500 && err.status !== 429) {
            throw err;
          }
        }
        if (attempt === maxRetries - 1) throw err;
        // Пауза 2 секунды перед повтором при 500 / 502 / network error
        await new Promise((r) => setTimeout(r, 2000));
      }
    }
    throw new ApiError(0, 'network', 'Не удалось загрузить файл после нескольких попыток');
  })();
}

// Клиентская копия правила именования (сервер всё равно проверяет повторно).

export const AUDIO_EXTENSIONS = ['.mp3', '.flac', '.wav', '.ogg', '.m4a', '.opus'];
export const NAMING_HINT = 'Переименуйте файл в формат «Исполнитель - Название.mp3» (дефис с пробелами по бокам).';

export function validateUploadName(fileName: string): { ok: true; artist: string; title: string } | { ok: false; error: string } {
  const dot = fileName.lastIndexOf('.');
  const ext = dot > 0 ? fileName.slice(dot).toLowerCase() : '';
  if (!AUDIO_EXTENSIONS.includes(ext)) {
    return { ok: false, error: `Неподдерживаемый формат. Допустимо: ${AUDIO_EXTENSIONS.join(', ')}` };
  }
  const base = fileName.slice(0, dot).replace(/\s+/g, ' ').trim();
  const idx = base.indexOf(' - ');
  const artist = idx > 0 ? base.slice(0, idx).trim() : '';
  const title = idx > 0 ? base.slice(idx + 3).trim() : '';
  if (!artist || !title) return { ok: false, error: `Имя файла не соответствует правилу. ${NAMING_HINT}` };
  return { ok: true, artist, title };
}

import { useEffect, useState } from 'react';

export function formatDuration(sec: number | undefined | null): string {
  if (!sec || !Number.isFinite(sec) || sec < 0) return '0:00';
  const s = Math.floor(sec);
  const h = Math.floor(s / 3600);
  const m = Math.floor((s % 3600) / 60);
  const r = s % 60;
  return h > 0 ? `${h}:${String(m).padStart(2, '0')}:${String(r).padStart(2, '0')}` : `${m}:${String(r).padStart(2, '0')}`;
}

export function formatBytes(bytes: number): string {
  if (!bytes) return '0 Б';
  const units = ['Б', 'КБ', 'МБ', 'ГБ', 'ТБ'];
  const i = Math.min(Math.floor(Math.log(bytes) / Math.log(1024)), units.length - 1);
  return `${(bytes / 1024 ** i).toFixed(i === 0 ? 0 : 1)} ${units[i]}`;
}

export function formatDate(iso: string): string {
  return new Date(iso).toLocaleDateString('ru-RU', { day: '2-digit', month: 'short', year: 'numeric' });
}

export function cn(...parts: (string | false | null | undefined)[]): string {
  return parts.filter(Boolean).join(' ');
}

export function useDebounced<T>(value: T, delay = 300): T {
  const [v, setV] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => setV(value), delay);
    return () => clearTimeout(t);
  }, [value, delay]);
  return v;
}

/** «2024-05-31» → начало/конец дня в локальной зоне, в ISO для API */
export function dayStartISO(date: string): string {

export function dayEndISO(date: string): string {

}

export function useAuthHydrated(): boolean {

'use client';

import { create } from 'zustand';
import { createJSONStorage, persist } from 'zustand/middleware';
import type { Track } from '@/lib/types';

export interface QueueItem {
  key: string;
  kind: 'local' | 'preview';
  /** id трека в нашей библиотеке (для kind = local) */
  id?: string;
  title: string;
  artist: string;
  coverUrl?: string | null;
  duration?: number;
  /** прямой URL (для превью) */
  src?: string;
}

export type NewQueueItem = Omit<QueueItem, 'key'>;
export type RepeatMode = 'off' | 'all' | 'one';

/** <audio> живёт в GlobalPlayer, а стор управляет им через эту ссылку. */
export const audioRef: { current: HTMLAudioElement | null } = { current: null };

let seq = 0;
const nextKey = () => `q${++seq}`;
const withKey = (i: NewQueueItem): QueueItem => ({ ...i, key: nextKey() });

export function trackToItem(t: Track): NewQueueItem {
  return { kind: 'local', id: t.id, title: t.title, artist: t.artist, coverUrl: t.coverUrl, duration: t.duration };
}

interface PlayerState {
  queue: QueueItem[];
  index: number;
  playing: boolean;
  shuffle: boolean;
  repeat: RepeatMode;
  volume: number;
  muted: boolean;
  currentTime: number;
  duration: number;
  /** индексы уже сыгранных треков (для shuffle и кнопки «назад») */
  history: number[];
  queueOpen: boolean;
  error: string | null;

  playQueue: (items: NewQueueItem[], start?: number) => void;
  toggle: () => void;
  next: (auto?: boolean) => void;
  prev: () => void;
  seek: (t: number) => void;
  setVolume: (v: number) => void;
  toggleMute: () => void;
  toggleShuffle: () => void;
  cycleRepeat: () => void;
  enqueue: (item: NewQueueItem) => void;
  playNext: (item: NewQueueItem) => void;
  removeAt: (i: number) => void;
  jumpTo: (i: number) => void;
  clear: () => void;
  setQueueOpen: (open: boolean) => void;
  // служебные — вызываются из GlobalPlayer
  _tick: (t: number, d: number) => void;
  _setPlaying: (p: boolean) => void;
  _setError: (e: string | null) => void;
}

const range = (n: number) => Array.from({ length: n }, (_, i) => i);

function restartCurrent() {
  const a = audioRef.current;
  if (a) {
    a.currentTime = 0;
    void a.play().catch(() => undefined);
  }
}

export const usePlayer = create<PlayerState>()(
  persist(
    (set, get) => {
      const pushHistory = (s: PlayerState, idx: number) => (s.shuffle ? [...s.history, idx] : [idx]);

      return {
        queue: [],
        index: 0,
        playing: false,
        shuffle: false,
        repeat: 'off',
        volume: 0.8,
        muted: false,
        currentTime: 0,
        duration: 0,
        history: [],
        queueOpen: false,
        error: null,

        playQueue: (items, start = 0) => {
          if (items.length === 0) return;
          const idx = Math.min(Math.max(start, 0), items.length - 1);
          set({ queue: items.map(withKey), index: idx, history: [idx], playing: true, currentTime: 0, duration: 0, error: null });
        },

        toggle: () => {
          const s = get();
          if (s.queue.length === 0) return;
          set({ playing: !s.playing, error: null });
        },

        next: (auto = false) => {
          const s = get();
          const n = s.queue.length;
          if (n === 0) return;
          if (auto && s.repeat === 'one') return restartCurrent();

          let nextIdx: number | null = null;
          let history = s.history;
          if (s.shuffle && n > 1) {
            const played = new Set(history);
            let candidates = range(n).filter((i) => !played.has(i));
            if (candidates.length === 0 && s.repeat === 'all') {
              history = [s.index]; // новый круг
              candidates = range(n).filter((i) => i !== s.index);
            }
            if (candidates.length) nextIdx = candidates[Math.floor(Math.random() * candidates.length)]!;
          } else if (s.index + 1 < n) {
            nextIdx = s.index + 1;
          } else if (s.repeat === 'all') {
            nextIdx = 0;
          }

          if (nextIdx === null) {
            if (auto) set({ playing: false, currentTime: 0 }); // очередь закончилась
            return;
          }
          if (nextIdx === s.index) return restartCurrent(); // в очереди один трек
          set({ index: nextIdx, history: pushHistory({ ...s, history }, nextIdx), playing: true, currentTime: 0, error: null });
        },

        prev: () => {
          const s = get();
          const a = audioRef.current;
          if (a && a.currentTime > 3) {
            a.currentTime = 0;
            return;
          }
          if (s.shuffle) {
            if (s.history.length > 1) {
              const h = s.history.slice(0, -1);
              set({ index: h[h.length - 1]!, history: h, playing: true, currentTime: 0, error: null });
            } else if (a) a.currentTime = 0;
            return;
          }
          const target = s.index > 0 ? s.index - 1 : s.repeat === 'all' ? s.queue.length - 1 : -1;
          if (target < 0 || target === s.index) {
            if (a) a.currentTime = 0;
            return;
          }
          set({ index: target, history: [target], playing: true, currentTime: 0, error: null });
        },

        seek: (t) => {
          const a = audioRef.current;
          if (a && Number.isFinite(t)) {
            a.currentTime = t;
            set({ currentTime: t });
          }
        },

        setVolume: (v) => set({ volume: Math.min(Math.max(v, 0), 1), muted: false }),
        toggleMute: () => set((s) => ({ muted: !s.muted })),
        toggleShuffle: () => set((s) => ({ shuffle: !s.shuffle, history: [s.index] })),
        cycleRepeat: () => set((s) => ({ repeat: s.repeat === 'off' ? 'all' : s.repeat === 'all' ? 'one' : 'off' })),

        enqueue: (item) => {
          const s = get();
          if (s.queue.length === 0) return get().playQueue([item]);
          set({ queue: [...s.queue, withKey(item)] });
        },

        playNext: (item) => {
          const s = get();
          if (s.queue.length === 0) return get().playQueue([item]);
          const at = s.index + 1;
          const queue = [...s.queue.slice(0, at), withKey(item), ...s.queue.slice(at)];
          set({ queue, history: s.history.map((h) => (h >= at ? h + 1 : h)) });
        },

        removeAt: (i) => {
          const s = get();
          const queue = s.queue.filter((_, k) => k !== i);
          if (queue.length === 0) return get().clear();
          let index = s.index;
          if (i < index) index -= 1;
          else if (i === index) index = Math.min(index, queue.length - 1); // продолжит следующий трек
          const history = s.history.filter((h) => h !== i).map((h) => (h > i ? h - 1 : h));
          set({ queue, index, history: history.length ? history : [index] });
        },

        jumpTo: (i) => {
          const s = get();
          if (i < 0 || i >= s.queue.length) return;
          set({ index: i, history: pushHistory(s, i), playing: true, currentTime: 0, error: null });
        },

        clear: () => {
          audioRef.current?.pause();
          set({ queue: [], index: 0, history: [], playing: false, currentTime: 0, duration: 0, error: null });
        },

        setQueueOpen: (queueOpen) => set({ queueOpen }),

        _tick: (currentTime, duration) => set({ currentTime, duration: Number.isFinite(duration) ? duration : 0 }),
        _setPlaying: (playing) => set({ playing }),
        _setError: (error) => set({ error, ...(error ? { playing: false } : {}) }),
      };
    },
    {
      name: 'ms-player',
      storage: createJSONStorage(() => localStorage),
      partialize: (s) => ({ volume: s.volume, shuffle: s.shuffle, repeat: s.repeat }),
    },
  ),
);
